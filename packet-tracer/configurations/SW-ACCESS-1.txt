!-------------------------------------------------------------------
! AegisMesh — Access Switch 1 Configuration
! Device: SW-ACCESS-1
! Model: Cisco 2960-24TT
! Role: Faculty workstations (VLAN 10) + DMZ server (VLAN 60)
! Traces to: SR-01, network-design.md Section 2.2
!-------------------------------------------------------------------

enable
configure terminal

!--- Hostname and Domain
hostname SW-ACCESS-1
ip domain-name aegismesh.local

!--- Security: Passwords and Access Control
enable secret AegisMesh2026!
service password-encryption
no ip domain-lookup
banner motd # UNAUTHORIZED ACCESS PROHIBITED - AegisMesh Access Switch 1 #

!--- Console Line Security
line console 0
 password ConsoleAccess2026!
 login
 logging synchronous
 exec-timeout 10 0
exit

!--- VTY Line Security
line vty 0 15
 password VTYAccess2026!
 login
 transport input ssh
 exec-timeout 10 0
exit

!===================================================================
! VLAN DEFINITIONS (must match SW-CORE)
!===================================================================

vlan 10
 name FACULTY
exit
vlan 20
 name APP-SERVERS
exit
vlan 30
 name MANAGEMENT
exit
vlan 40
 name DATABASE
exit
vlan 50
 name SECURITY
exit
vlan 60
 name DMZ
exit
vlan 99
 name NATIVE-UNUSED
exit

!===================================================================
! ACCESS PORTS — Faculty Workstations (VLAN 10)
!===================================================================

interface FastEthernet0/1
 description FAC-PC-01
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
exit

interface FastEthernet0/2
 description FAC-PC-02
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
exit

interface FastEthernet0/3
 description FAC-PC-03
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
exit

!===================================================================
! ACCESS PORT — DMZ Server (VLAN 60)
!===================================================================

interface FastEthernet0/10
 description DMZ-SRV-01
 switchport mode access
 switchport access vlan 60
 spanning-tree portfast
 no shutdown
exit

!===================================================================
! TRUNK PORT — Uplink to SW-CORE
!===================================================================

interface FastEthernet0/24
 description TRUNK-TO-SW-CORE
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60
 switchport nonegotiate
 no shutdown
exit

!===================================================================
! SHUTDOWN UNUSED PORTS (Hardening)
! Assign to blackhole VLAN 99 and shut down
!===================================================================

interface range FastEthernet0/4-9
 switchport mode access
 switchport access vlan 99
 shutdown
exit

interface range FastEthernet0/11-23
 switchport mode access
 switchport access vlan 99
 shutdown
exit

interface GigabitEthernet0/1
 shutdown
exit

interface GigabitEthernet0/2
 shutdown
exit

!--- Save Configuration
end
write memory
