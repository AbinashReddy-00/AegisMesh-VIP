!-------------------------------------------------------------------
! AegisMesh — Core Multilayer Switch Configuration
! Device: SW-CORE
! Model: Cisco 3560-24PS
! Role: Inter-VLAN routing, trunk aggregation, DHCP, ACL enforcement
! Traces to: SR-01 (network segmentation), network-design.md Sections 3–6
!
! CRITICAL DESIGN NOTE — ACL Return Traffic:
! ------------------------------------------
! ACLs on SVIs are STATELESS in Packet Tracer. When we apply an inbound
! ACL on a VLAN SVI, it filters traffic ORIGINATING from that VLAN.
! For bidirectional communication (e.g., Faculty pings App Server),
! the ECHO REQUEST exits Faculty VLAN (checked by FACULTY-ACCESS) and
! the ECHO REPLY exits App VLAN (checked by APP-SERVER-ACCESS).
!
! Therefore, ACLs must explicitly permit RETURN traffic for legitimate
! flows. For example:
!   - FACULTY-ACCESS permits Faculty → App (outbound request)
!   - APP-SERVER-ACCESS permits App → Faculty (return response)
!
! This is NOT a security weakness — it allows bidirectional communication
! only between explicitly authorized zone pairs. The security boundaries
! (Faculty ✗ Database, DMZ ✗ Management, etc.) remain intact because
! neither direction is permitted for those pairs.
!-------------------------------------------------------------------

enable
configure terminal

!--- Hostname and Domain
hostname SW-CORE
ip domain-name aegismesh.local

!--- Security: Passwords and Access Control
enable secret AegisMesh2026!
service password-encryption
no ip domain-lookup
banner motd # UNAUTHORIZED ACCESS PROHIBITED - AegisMesh Core Switch #

!--- Console Line Security
line console 0
 password ConsoleAccess2026!
 login
 logging synchronous
 exec-timeout 10 0
exit

!--- VTY Line Security (SSH only, restricted to Management VLAN)
username admin privilege 15 secret AdminSSH2026!
crypto key generate rsa general-keys modulus 1024
ip ssh version 2

line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
 access-class MGMT-VTY-ACCESS in
exit

ip access-list standard MGMT-VTY-ACCESS
 permit 10.10.30.0 0.0.0.255
 deny any
exit

!--- Enable IP Routing (required for inter-VLAN routing)
ip routing

!===================================================================

!===================================================================

ip dhcp excluded-address 10.10.10.1 10.10.10.99

!===================================================================
! ROUTING
!===================================================================

!--- Default route to Edge Router (for internet-bound traffic)
ip route 0.0.0.0 0.0.0.0 10.10.0.1

!===================================================================
! ACCESS CONTROL LISTS
!===================================================================
!
! ACL STRATEGY:
! - Applied INBOUND on each VLAN SVI
! - Controls traffic ORIGINATING from that VLAN
! - Must account for return traffic (stateless ACLs)
! - Default action: implicit deny all (Cisco default)
!
! Return traffic is explicitly permitted only between
! authorized zone pairs. Unauthorized pairs are blocked
! in BOTH directions.
!===================================================================

!-------------------------------------------------------------------
! ACL: FACULTY-ACCESS
! Applied to: interface Vlan10, direction: in
! Purpose: Control what Faculty VLAN (10) can initiate/respond to
!
! Faculty CAN: access App Servers, access DMZ, reach Internet
! Faculty CANNOT: access Database, Management, Security/Logging
!-------------------------------------------------------------------

ip access-list extended FACULTY-ACCESS
 remark --- PERMIT: Faculty to App Servers (authorized access) ---
 permit ip 10.10.10.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark --- PERMIT: Faculty to DMZ (authorized access) ---
 permit ip 10.10.10.0 0.0.0.255 10.10.60.0 0.0.0.255
 remark --- DENY: Faculty to Management (lateral movement prevention) ---
 deny ip 10.10.10.0 0.0.0.255 10.10.30.0 0.0.0.255
 remark --- DENY: Faculty to Database (must go through app layer) ---
 deny ip 10.10.10.0 0.0.0.255 10.10.40.0 0.0.0.255
 remark --- DENY: Faculty to Security/Logging (privileged zone) ---
 deny ip 10.10.10.0 0.0.0.255 10.10.50.0 0.0.0.255
 remark --- PERMIT: Faculty to Internet (via default route) ---
 permit ip 10.10.10.0 0.0.0.255 any
exit

!-------------------------------------------------------------------
! ACL: APP-SERVER-ACCESS
! Applied to: interface Vlan20, direction: in
! Purpose: Control what App Server VLAN (20) can initiate/respond to
!
! App Servers CAN: access Database, Logging, respond to Faculty/DMZ
! App Servers CANNOT: access Management
!
! DESIGN NOTE: App → Faculty and App → DMZ are permitted because
! App Servers must send RESPONSES to Faculty/DMZ requests.
! Packet Tracer ACLs are stateless — without these rules,
! return traffic (e.g., HTTP responses, ICMP echo-reply) would
! be dropped at the VLAN 20 SVI, breaking legitimate communication.
!-------------------------------------------------------------------

ip access-list extended APP-SERVER-ACCESS
 remark --- PERMIT: App to Database (authorized data access) ---
 permit ip 10.10.20.0 0.0.0.255 10.10.40.0 0.0.0.255
 remark --- PERMIT: App to Security/Logging (log forwarding) ---
 permit ip 10.10.20.0 0.0.0.255 10.10.50.0 0.0.0.255
 remark --- PERMIT: App to Faculty (response traffic) ---
 permit ip 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255
 remark --- PERMIT: App to DMZ (response traffic) ---
 permit ip 10.10.20.0 0.0.0.255 10.10.60.0 0.0.0.255
 remark --- DENY: App to Management (lateral movement prevention) ---
 deny ip 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255
 remark --- DENY: App to all other (default deny) ---
 deny ip any any
exit

!-------------------------------------------------------------------
! ACL: DMZ-ACCESS
! Applied to: interface Vlan60, direction: in
! Purpose: Control what DMZ VLAN (60) can initiate/respond to
!
! DMZ CAN: access App Servers (reverse proxy), Logging
! DMZ CANNOT: access Database, Management, Faculty, any other
!-------------------------------------------------------------------

ip access-list extended DMZ-ACCESS
 remark --- PERMIT: DMZ to App Servers (reverse proxy path) ---
 permit ip 10.10.60.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark --- PERMIT: DMZ to Security/Logging (log forwarding) ---
 permit ip 10.10.60.0 0.0.0.255 10.10.50.0 0.0.0.255
 remark --- PERMIT: DMZ to Faculty (stateless return traffic for Faculty-initiated requests) ---
 permit ip 10.10.60.0 0.0.0.255 10.10.10.0 0.0.0.255
 remark --- DENY: DMZ to Database (critical data protection) ---
 deny ip 10.10.60.0 0.0.0.255 10.10.40.0 0.0.0.255
 remark --- DENY: DMZ to Management (privilege escalation prevention) ---
 deny ip 10.10.60.0 0.0.0.255 10.10.30.0 0.0.0.255
 remark --- DENY: DMZ to all other (default deny) ---
 deny ip any any
exit

!-------------------------------------------------------------------
! ACL: MGMT-ACCESS
! Applied to: interface Vlan30, direction: in
! Purpose: Control what Management VLAN (30) can initiate/respond to
!
! Management CAN: access App Servers, Database, Logging (admin duties)
! Management CANNOT: access Faculty, DMZ, Internet
!
! Management is a HIGH-security zone. Admin access to App and DB
! is permitted for infrastructure management. No outbound internet.
!-------------------------------------------------------------------

ip access-list extended MGMT-ACCESS
 remark --- PERMIT: Mgmt to App Servers (administration) ---
 permit ip 10.10.30.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark --- PERMIT: Mgmt to Database (administration) ---
 permit ip 10.10.30.0 0.0.0.255 10.10.40.0 0.0.0.255
 remark --- PERMIT: Mgmt to Security/Logging (log review) ---
 permit ip 10.10.30.0 0.0.0.255 10.10.50.0 0.0.0.255
 remark --- DENY: Mgmt to all other (restricted zone) ---
 deny ip any any
exit

!-------------------------------------------------------------------
! ACL: DB-ACCESS
! Applied to: interface Vlan40, direction: in
! Purpose: Control what Database VLAN (40) can initiate/respond to
!
! Database CAN: send logs to Security, respond to App Servers/Mgmt
! Database CANNOT: initiate connections to Faculty, DMZ, Internet
!
! DESIGN NOTE: DB → App and DB → Mgmt are permitted for RETURN
! traffic. Without these, App → DB queries would succeed outbound
! but the DB response would be dropped at the VLAN 40 SVI.
!-------------------------------------------------------------------

ip access-list extended DB-ACCESS
 remark --- PERMIT: DB to Security/Logging (log forwarding) ---
 permit ip 10.10.40.0 0.0.0.255 10.10.50.0 0.0.0.255
 remark --- PERMIT: DB to App Servers (response traffic) ---
 permit ip 10.10.40.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark --- PERMIT: DB to Management (response traffic) ---
 permit ip 10.10.40.0 0.0.0.255 10.10.30.0 0.0.0.255
 remark --- DENY: DB to all other (default deny) ---
 deny ip any any
exit

!-------------------------------------------------------------------
! ACL: SEC-ACCESS
! Applied to: interface Vlan50, direction: in
! Purpose: Control what Security/Logging zone (VLAN 50) can initiate
!
! Hardened Least-Privilege Policy:
! Security CAN: reach App, DB, DMZ, and Management for telemetry & polling
! Security CANNOT: initiate traffic to Faculty PCs or External Internet
!-------------------------------------------------------------------

ip access-list extended SEC-ACCESS
 remark --- PERMIT: SIEM/Monitoring to App Servers ---
 permit ip 10.10.50.0 0.0.0.255 10.10.20.0 0.0.0.255
 remark --- PERMIT: SIEM/Monitoring to Database Servers ---
 permit ip 10.10.50.0 0.0.0.255 10.10.40.0 0.0.0.255
 remark --- PERMIT: SIEM/Monitoring to DMZ Servers ---
 permit ip 10.10.50.0 0.0.0.255 10.10.60.0 0.0.0.255
 remark --- PERMIT: SIEM/Monitoring to Management VLAN ---
 permit ip 10.10.50.0 0.0.0.255 10.10.30.0 0.0.0.255
 remark --- DENY: SIEM to Faculty PCs ---
 deny ip 10.10.50.0 0.0.0.255 10.10.10.0 0.0.0.255
 remark --- DENY: SIEM to Internet (Exfiltration prevention) ---
 deny ip 10.10.50.0 0.0.0.255 any
 remark --- DENY: Default Deny ---
 deny ip any any
exit

!===================================================================
! LOGGING
!===================================================================

logging 10.10.50.10

!--- Save Configuration
end
write memory
