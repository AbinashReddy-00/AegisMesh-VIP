!-------------------------------------------------------------------
! AegisMesh — Access Switch 2 Configuration
! Device: SW-ACCESS-2
! Model: Cisco 2960-24TT
! Role: Application Servers (VLAN 20) + Database Servers (VLAN 40)
! Traces to: SR-01, network-design.md Section 2.2
!-------------------------------------------------------------------

enable
configure terminal

!--- Hostname and Domain
hostname SW-ACCESS-2
ip domain-name aegismesh.local

!--- Security: Passwords and Access Control
enable secret AegisMesh2026!
service password-encryption
no ip domain-lookup
banner motd # UNAUTHORIZED ACCESS PROHIBITED - AegisMesh Access Switch 2 #

!--- Console Line Security
line console 0
 password ConsoleAccess2026!
 login
 logging synchronous
 exec-timeout 10 0
exit

!--- VTY Line Security
line vty 0 15
 password VTYAccess2026!
 login
 transport input ssh
 exec-timeout 10 0
exit

!===================================================================
! VLAN DEFINITIONS (must match SW-CORE)
!===================================================================

vlan 10
 name FACULTY
exit
vlan 20
 name APP-SERVERS
exit
vlan 30
 name MANAGEMENT
exit
vlan 40
 name DATABASE
exit
vlan 50
 name SECURITY
exit
vlan 60
 name DMZ
exit
vlan 99
 name NATIVE-UNUSED
exit

!===================================================================
! ACCESS PORTS — Application Servers (VLAN 20)
!===================================================================

interface FastEthernet0/1
 description APP-SRV-01-EDUCATION
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit

interface FastEthernet0/2
 description APP-SRV-02-RESEARCH
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
exit

!===================================================================
! ACCESS PORTS — Database Servers (VLAN 40)
!===================================================================

interface FastEthernet0/3
 description DB-SRV-01-PRIMARY
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown
exit

interface FastEthernet0/4
 description DB-SRV-02-SECONDARY
 switchport mode access
 switchport access vlan 40
 spanning-tree portfast
 no shutdown
exit

!===================================================================
! TRUNK PORT — Uplink to SW-CORE
!===================================================================

interface FastEthernet0/24
 description TRUNK-TO-SW-CORE
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,50,60
 switchport nonegotiate
 no shutdown
exit

!===================================================================
! SHUTDOWN UNUSED PORTS (Hardening)
!===================================================================

interface range FastEthernet0/5-23
 switchport mode access
 switchport access vlan 99
 shutdown
exit

interface GigabitEthernet0/1
 shutdown
exit

interface GigabitEthernet0/2
 shutdown
exit

!--- Save Configuration
end
write memory
